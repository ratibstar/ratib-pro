# === PASTE THIS BLOCK into public_html/.htaccess (replace the old Profile rules) ===
# Company profile — serve about.php directly (no redirect loop).
RewriteRule ^profile/?$ /pages/about.php [L,QSA]
RewriteRule ^pages/profile/?$ /profile [R=302,L]

# Legacy URLs → /profile (THE_REQUEST = browser URL only).
RewriteCond %{THE_REQUEST} \s/+pages/company-profile\.php[\s?] [NC]
RewriteRule ^pages/company-profile\.php$ /profile [R=302,L,QSA]
RewriteCond %{THE_REQUEST} \s/+pages/home\.php[\s?] [NC]
RewriteCond %{QUERY_STRING} (^|&)(open=profile|open=about)(&|$) [NC]
RewriteRule ^pages/home\.php$ /profile [R=302,L]
# === END Profile block ===
