⚙️ أوامر مفيدة:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# التحقق من حالة الخوادم:
tasklist | find "mysqld"   # MySQL
tasklist | find "httpd"    # Apache

# إيقاف الخوادم:
taskkill /F /IM mysqld.exe  # MySQL
taskkill /F /IM httpd.exe   # Apache

# تشغيل الخوادم مجدداً:
cd C:\xampp\mysql\bin && mysqld.exe
cd C:\xampp\apache\bin && httpd.exe
